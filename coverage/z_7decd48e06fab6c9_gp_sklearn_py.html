3from sklearn.gaussian_process import GaussianProcessRegressor 

4from sklearn.gaussian_process.kernels import RBF, ConstantKernel, Matern, WhiteKernel

8def default_gp_kernel(): 

9 """Amplitude * RBF + learned noise. 

10 

11 sklearn's default (fixed amplitude 1, no noise term, alpha=1e-10) has to 

12 interpolate noisy data and collapses to a tiny length scale. 

13 """ 

14 return ConstantKernel(1.0) * RBF(1.0) + WhiteKernel(1e-2)

16 

17def matern_gp_kernel(): 

18 """Amplitude * Matern(nu=2.5) + learned noise.""" 

19 return ConstantKernel(1.0) * Matern(1.0, nu=2.5) + WhiteKernel(1e-2)

22class GaussianProcessSurrogate(BaseSurrogate): 

23 """Gaussian Process surrogate with uncertainty. 

24 

25 Defaults to :func:`default_gp_kernel`, ``normalize_y=True`` and standardised 

26 inputs; any ``GaussianProcessRegressor`` keyword overrides them. 

27 """ 

28 def __init__(self, scale_inputs=True, **kwargs): 

29 kwargs.setdefault("kernel", default_gp_kernel()) 

30 kwargs.setdefault("normalize_y", True) 

31 super().__init__(GaussianProcessRegressor(**kwargs), scale_inputs=scale_inputs)
