3"""Save and load fitted surrogates together with what is needed to reuse them. 

4 

5A fitted model on its own is not enough to reuse it later: you also need the 

6input columns it expects (in order), the names of its outputs, the fidelity 

7levels of a multi-fidelity model, and the package versions it was fitted 

8with (a pickle from one scikit-learn version may not load, or may predict 

9differently, under another). :func:`save_model` stores all of these in one 

10``joblib`` file and :func:`load_model` returns them as a :class:`ModelBundle`. 

11 

12The file layout uses the same keys as the Grid Search app's 

13"Download Trained Model" bundle (``input_columns``, ``output_columns``, 

14``versions``), and :func:`load_model` also opens those files and plain 

15``joblib.dump(model)`` files. 

16 

17.. warning:: 

18 ``joblib`` files are pickles: loading one runs code from it. Only load 

19 files you created or trust. 

20""" 

21 

22from __future__ import annotations 

23 

24import sys 

25import warnings 

26from dataclasses import dataclass, field 

27from datetime import datetime, timezone 

28from importlib.metadata import PackageNotFoundError, version 

29from typing import Any, Dict, List, Optional, Sequence 

30 

31import joblib 

32import numpy as np 

33 

34FORMAT = "multioutreg.model_bundle" 

35FORMAT_VERSION = 1 

36 

37# Versions recorded at save time and compared at load time. 

38_CHECKED_PACKAGES = ("multioutreg", "scikit-learn", "numpy") 

39_RECORDED_PACKAGES = _CHECKED_PACKAGES + ("joblib", "torch", "gpytorch") 

40 

41 

42def package_versions() -> Dict[str, str]: 

43 """Installed versions of Python and the packages a saved model depends on.""" 

44 versions = {"python": sys.version.split()[0]} 

45 for pkg in _RECORDED_PACKAGES: 

46 try: 

47 versions[pkg] = version(pkg) 

48 except PackageNotFoundError: 

49 versions[pkg] = "not installed" 

50 return versions 

51 

52 

53@dataclass 

54class ModelBundle: 

55 """A fitted model plus the information needed to use it. 

56 

57 ``predict`` accepts a DataFrame (columns are selected and ordered by 

58 ``input_columns``) or an array (used as is), and passes keyword arguments 

59 such as ``return_std=True`` through to the model. 

60 """ 

61 

62 model: Any 

63 input_columns: Optional[List[str]] = None 

64 output_columns: Optional[List[str]] = None 

65 fidelity_levels: Optional[List[str]] = None 

66 versions: Dict[str, str] = field(default_factory=dict) 

67 metadata: Dict[str, Any] = field(default_factory=dict) 

68 conformal: Optional[Dict[str, Any]] = None 

69 

70 def _prepare(self, X): 

71 if hasattr(X, "columns") and self.input_columns is not None: 

72 missing = [c for c in self.input_columns if c not in X.columns] 

73 if missing: 

74 raise KeyError(f"input columns missing from X: {missing}") 

75 return X[self.input_columns].to_numpy() 

76 return X 

77 

78 def predict(self, X, as_frame: bool = False, **kwargs): 

79 """Predict with the stored model. 

80 

81 With ``as_frame=True`` every returned array is a DataFrame with 

82 ``output_columns`` as its columns (requires pandas and output names). 

83 """ 

84 out = self.model.predict(self._prepare(X), **kwargs) 

85 if not as_frame: 

86 return out 

87 import pandas as pd 

88 

89 index = X.index if hasattr(X, "index") else None 

90 cols = self.output_columns 

91 

92 def frame(a): 

93 a = np.asarray(a) 

94 return pd.DataFrame(a.reshape(len(a), -1), columns=cols, index=index) 

95 

96 return tuple(frame(a) for a in out) if isinstance(out, tuple) else frame(out)

98 

99def _infer_input_columns(model): 

100 names = getattr(model, "feature_names_in_", None) 

101 return None if names is None else [str(n) for n in names] 

102

104def save_model( 

105 model: Any, 

106 path, 

107 input_columns: Optional[Sequence[str]] = None, 

108 output_columns: Optional[Sequence[str]] = None, 

109 fidelity_levels: Optional[Sequence[str]] = None, 

110 metadata: Optional[Dict[str, Any]] = None, 

111 compress: int = 3, 

112): 

113 """Save a fitted model with its column names, fidelity levels and versions. 

114 

115 Parameters 

116 ---------- 

117 model : object 

118 Any fitted surrogate, pipeline or estimator. 

119 path : str | os.PathLike | file-like 

120 Destination (``.joblib`` by convention). 

121 input_columns, output_columns : sequence of str, optional 

122 Feature and output names. ``input_columns`` defaults to the model's 

123 ``feature_names_in_`` when it has one. 

124 fidelity_levels : sequence of str, optional 

125 Defaults to the model's ``fidelity_levels`` attribute when it has one 

126 (``MultiFidelitySurrogate``, ``StackedVFMSurrogate``). 

127 metadata : dict, optional 

128 Anything else to keep with the model (training data source, metrics, 

129 the configuration that produced it). 

130 compress : int 

131 ``joblib`` compression level, 0-9. 

132 

133 Returns 

134 ------- 

135 The list of file names written, as returned by ``joblib.dump``. 

136 """ 

137 if input_columns is None: 

138 input_columns = _infer_input_columns(model) 

139 if fidelity_levels is None: 

140 fidelity_levels = getattr(model, "fidelity_levels", None) 

141 bundle = { 

142 "format": FORMAT, 

143 "format_version": FORMAT_VERSION, 

144 "model": model, 

145 "model_class": f"{type(model).__module__}.{type(model).__qualname__}", 

146 "input_columns": None if input_columns is None else [str(c) for c in input_columns], 

147 "output_columns": None if output_columns is None else [str(c) for c in output_columns], 

148 "fidelity_levels": None if fidelity_levels is None else [str(f) for f in fidelity_levels], 

149 "versions": package_versions(), 

150 "saved_at": datetime.now(timezone.utc).isoformat(timespec="seconds"), 

151 "metadata": dict(metadata or {}), 

152 } 

153 return joblib.dump(bundle, path, compress=compress)

156def _warn_on_version_mismatch(saved: Dict[str, str]): 

157 current = package_versions() 

158 diffs = [ 

159 f"{pkg} {saved[pkg]} (now {current[pkg]})" 

160 for pkg in _CHECKED_PACKAGES 

161 if pkg in saved and saved[pkg] != current.get(pkg) 

162 ] 

163 if diffs: 

164 warnings.warn( 

165 "Model was saved with different package versions: " + ", ".join(diffs) 

166 + ". Predictions may differ; refit if possible.", 

167 UserWarning, 

168 stacklevel=3, 

169 )

172def load_model(path, check_versions: bool = True) -> ModelBundle: 

173 """Load a model saved by :func:`save_model`, the Grid Search app, or ``joblib.dump``. 

174 

175 Only load files you trust: ``joblib`` files are pickles. 

176 

177 Parameters 

178 ---------- 

179 path : str | os.PathLike | file-like 

180 check_versions : bool 

181 Warn if ``multioutreg``, scikit-learn or numpy differ from the 

182 versions the model was saved with. 

183 """ 

184 obj = joblib.load(path) 

185 if isinstance(obj, dict) and obj.get("format") == FORMAT: 

186 if obj.get("format_version", 0) > FORMAT_VERSION: 

187 raise ValueError( 

188 f"{path!r} uses bundle format {obj['format_version']}, newer than this " 

189 f"multioutreg supports ({FORMAT_VERSION}); upgrade multioutreg." 

190 ) 

191 bundle = ModelBundle( 

192 model=obj["model"], 

193 input_columns=obj.get("input_columns"), 

194 output_columns=obj.get("output_columns"), 

195 fidelity_levels=obj.get("fidelity_levels"), 

196 versions=obj.get("versions", {}), 

197 metadata={**obj.get("metadata", {}), "saved_at": obj.get("saved_at")}, 

198 conformal=obj.get("metadata", {}).get("conformal"), 

199 ) 

200 elif isinstance(obj, dict) and "pipeline" in obj: 

201 # Grid Search app download (multioutreg.gui.grid_search_core.build_model_bundle). 

202 bundle = ModelBundle( 

203 model=obj["pipeline"], 

204 input_columns=obj.get("input_columns"), 

205 output_columns=obj.get("output_columns"), 

206 versions=obj.get("versions", {}), 

207 metadata={"config": obj.get("config")}, 

208 conformal=obj.get("conformal"), 

209 ) 

210 else: 

211 bundle = ModelBundle(model=obj) 

212 if check_versions and bundle.versions: 

213 _warn_on_version_mismatch(bundle.versions) 

214 return bundle
