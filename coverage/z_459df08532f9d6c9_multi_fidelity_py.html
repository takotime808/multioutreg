2"""Cost-aware multi-fidelity active learning. 

3 

4Each query is a pair ``(x, level)``. The acquisitions estimate how much a run 

5at each level would reduce uncertainty about the *highest* fidelity, divided 

6by its cost. See the "Multi-Fidelity Active Learning" section of 

7``docs/active_learning.md``. 

8"""

29 """High-fidelity prediction at candidates, split by what each level can reduce.

48def _is_ar1(model) -> bool: 

49 return all(hasattr(model, a) for a in ("surrogate_lo_", "surrogate_delta_", "rho_"))

52def _column_corr(A, B) -> np.ndarray: 

53 """Per-column Pearson correlation (0 where a column is constant).""" 

54 A, B = _as_2d(A), _as_2d(B) 

55 out = np.zeros(A.shape[1]) 

56 for j in range(A.shape[1]): 

57 if A[:, j].std() > 0 and B[:, j].std() > 0: 

58 out[j] = np.corrcoef(A[:, j], B[:, j])[0, 1] 

59 return np.nan_to_num(out)

62def _predict_level(model, X, level, levels): 

63 if _is_ar1(model): 

64 return _as_2d(model.surrogate_lo_.predict(X)) if level == levels[0] else _as_2d(model.predict(X)) 

65 return _as_2d(model.predict(X, level=level))

68def level_trust(model, levels: Sequence[str], X_hi, Y_hi, z: float = 1.0, min_points: int = 6) -> Dict[str, np.ndarray]: 

69 """How far the high-fidelity data supports each lower level, per output, in [0, 1]. 

70 

71 The squared lower confidence bound (Fisher z, ``z`` standard errors) of 

72 ``|corr(level prediction, Y_hi)|`` at the high-fidelity points. It is 0 

73 until there are ``min_points`` high-fidelity points (with 4 points, pure 

74 noise reaches ``|r| > 0.95`` a few percent of the time), so a lower level earns 

75 credit only once the data shows it predicts the high fidelity. Without it, 

76 a scale such as AR1's ``rho`` fitted to a handful of points can make an 

77 uninformative level look valuable indefinitely. 

78 """ 

79 X_hi, Y_hi = np.asarray(X_hi, dtype=np.float64), _as_2d(Y_hi) 

80 n = len(X_hi) 

81 trust = {} 

82 for lvl in list(levels)[:-1]: 

83 if n < max(min_points, 4): 

84 trust[lvl] = np.zeros(Y_hi.shape[1]) 

85 continue 

86 r = np.clip(np.abs(_column_corr(_predict_level(model, X_hi, lvl, levels), Y_hi)), 0.0, 1 - 1e-12) 

87 lower = np.tanh(np.arctanh(r) - z / np.sqrt(n - 3)) 

88 trust[lvl] = np.maximum(lower, 0.0) ** 2 

89 return trust

92def fidelity_info(model, X, levels: Sequence[str], nested: bool = True, trust: Optional[Dict[str, np.ndarray]] = None) -> FidelityInfo: 

93 """Per-level variance reduction for the high-fidelity prediction at ``X``. 

94 

95 For AR1 models (``AdditiveCorrectionVFM``), 

96 ``var_hi = rho^2 var_lo + var_delta``: a low-fidelity run removes the first 

97 term; a high-fidelity run removes ``var_delta``, plus the first term when 

98 ``nested`` (the low fidelity is run at the same point too). 

99 

100 For other models with ``predict(X, level=..., return_std=True)`` the 

101 reduction is ``corr_l^2 var_hi``, with ``corr_l`` the correlation of level 

102 ``l`` and high-fidelity predictions over ``X`` (Huang et al. 2006). 

103 

104 ``trust`` (see :func:`level_trust`) multiplies each lower level's 

105 reduction, and its square root that level's correlation. 

106 """ 

107 X = np.asarray(X, dtype=np.float64) 

108 levels = list(levels) 

109 top = levels[-1] 

110 if _is_ar1(model): 

111 if len(levels) != 2: 

112 raise ValueError("AR1 models have exactly two levels.") 

113 _, s_lo = model.surrogate_lo_.predict(X, return_std=True) 

114 _, s_d = model.surrogate_delta_.predict(X, return_std=True) 

115 mean, std = model.predict(X, return_std=True) 

116 mean, std = _as_2d(mean), _as_2d(std) 

117 v_lo = (model.rho_ * _as_2d(s_lo)) ** 2 

118 v_d = _as_2d(s_d) ** 2 

119 with np.errstate(divide="ignore", invalid="ignore"): 

120 c_lo = np.where(std > 0, np.sqrt(v_lo) / std, 0.0) 

121 reduction = {levels[0]: v_lo, top: v_d + (v_lo if nested else 0.0)} 

122 corr = {levels[0]: c_lo, top: np.ones_like(std)} 

123 else: 

124 mean, std = model.predict(X, level=top, return_std=True) 

125 mean, std = _as_2d(mean), _as_2d(std) 

126 reduction, corr = {}, {} 

127 for lvl in levels[:-1]: 

128 m_l = _predict_level(model, X, lvl, levels) 

129 if m_l.shape != mean.shape: 

130 raise ValueError(f"Level {lvl!r} has {m_l.shape[1]} outputs, high fidelity has {mean.shape[1]}.") 

131 c = np.clip(_column_corr(m_l, mean), 0.0, 1.0) 

132 corr[lvl] = np.broadcast_to(c, mean.shape).copy() 

133 reduction[lvl] = corr[lvl] ** 2 * std ** 2 

134 reduction[top], corr[top] = std ** 2, np.ones_like(std) 

135 if trust is not None: 

136 for lvl, t in trust.items(): 

137 reduction[lvl] = reduction[lvl] * t 

138 corr[lvl] = corr[lvl] * np.sqrt(t) 

139 return FidelityInfo(mean, std, reduction, corr)

142def effective_costs(levels: Sequence[str], costs: Dict[str, float], nested: bool = True) -> Dict[str, float]: 

143 """Cost of one query per level; nested queries also pay for every lower level.""" 

144 out, running = {}, 0.0 

145 for lvl in levels: 

146 running += costs[lvl] 

147 out[lvl] = running if nested else costs[lvl] 

148 return out

151def variance_reduction_per_cost(info: FidelityInfo, eff_costs: Dict[str, float], y_scale=None, how: str = "sum", weights=None) -> np.ndarray: 

152 """Standardized high-fidelity variance removed per unit cost, shape (n, n_levels).""" 

153 cols = [] 

154 for lvl, red in info.reduction.items(): 

155 r = red if y_scale is None else red / np.asarray(y_scale) ** 2 

156 cols.append(aggregate(r, how=how, weights=weights) / eff_costs[lvl]) 

157 return np.column_stack(cols)

160def mf_expected_improvement(info: FidelityInfo, eff_costs: Dict[str, float], best: float, xi: float = 0.0, weights=None, minimize: bool = True) -> np.ndarray: 

161 """Multi-fidelity EI (Huang et al. 2006): ``EI_hi * corr_l * C_top / C_l``, shape (n, n_levels).""" 

162 ei = expected_improvement(info.mean, info.std, best, xi=xi, weights=weights, minimize=minimize) 

163 _, sigma = _scalarize(info.mean, info.std, weights) 

164 top_cost = list(eff_costs.values())[-1] 

165 cols = [] 

166 for lvl, c in info.corr.items(): 

167 # Correlation of the scalarized objective: variance-weighted mean over outputs. 

168 _, s_l = _scalarize(info.mean, c * info.std, weights) 

169 with np.errstate(divide="ignore", invalid="ignore"): 

170 c_g = np.where(sigma > 0, s_l / sigma, 0.0) 

171 cols.append(ei * c_g * top_cost / eff_costs[lvl]) 

172 return np.column_stack(cols)

175class MultiFidelityActiveLearner: 

176 """Choose the next ``(x, level)`` to run under a total cost budget. 

177 

178 Parameters 

179 ---------- 

180 model_factory : callable 

181 Zero-argument callable returning an unfitted multi-fidelity surrogate 

182 with ``fit({level: (X, Y)})``, e.g. ``AdditiveCorrectionVFM`` (two 

183 levels named ``"lo"`` / ``"hi"``) or ``StackedVFMSurrogate``. 

184 bounds : array-like, shape (n_features, 2) 

185 levels : sequence of str 

186 Fidelity levels, cheapest first; the last one is the target. 

187 costs : dict[level, float] 

188 Cost of one evaluation at each level. 

189 acquisition : {"variance", "ei"}, default "variance" 

190 ``"variance"``: variance reduction per cost (global accuracy). 

191 ``"ei"``: multi-fidelity expected improvement (optimization). 

192 nested : bool, default True 

193 A query at a level also runs every lower level at the same point. 

194 min_hi_fraction : float, default 0.25 

195 Force a high-fidelity query whenever high-fidelity spending is below 

196 this share of the total spent. Evidence about whether a lower level 

197 is informative only comes from high-fidelity runs, so this keeps it 

198 coming even when a lower level (spuriously) looks valuable. 

199 trust_z : float or None, default 1.0 

200 Standard errors for :func:`level_trust`; lower levels are credited 

201 only as far as the high-fidelity data shows they predict it. ``None`` 

202 trusts the model's own variance split. 

203 """ 

204 

205 def __init__( 

206 self, 

207 model_factory: Callable, 

208 bounds, 

209 levels: Sequence[str], 

210 costs: Dict[str, float], 

211 acquisition: str = "variance", 

212 nested: bool = True, 

213 min_hi_fraction: float = 0.25, 

214 trust_z: Optional[float] = 1.0, 

215 how: str = "sum", 

216 weights=None, 

217 minimize: bool = True, 

218 xi: float = 0.0, 

219 n_candidates: int = 1024, 

220 random_state: Optional[int] = 0, 

221 ): 

222 if acquisition not in ("variance", "ei"): 

223 raise ValueError("acquisition must be 'variance' or 'ei'.") 

224 missing = set(levels) - set(costs) 

225 if missing: 

226 raise ValueError(f"No cost given for level(s) {sorted(missing)}.") 

227 self.model_factory = model_factory 

228 self.levels = list(levels) 

229 self.costs = {lvl: float(costs[lvl]) for lvl in self.levels} 

230 self.acquisition = acquisition 

231 self.nested = nested 

232 self.min_hi_fraction = min_hi_fraction 

233 self.trust_z = trust_z 

234 self.how = how 

235 self.weights = weights 

236 self.minimize = minimize 

237 self.xi = xi 

238 self.n_candidates = n_candidates 

239 # Reuse the single-fidelity learner for coordinates and seeding. 

240 self._space = ActiveLearner(lambda: None, bounds, random_state=random_state) 

241 self.data_: Dict[str, List[np.ndarray]] = {} 

242 self.spent_ = {lvl: 0.0 for lvl in self.levels} 

243 self.model_ = None 

244 self.history_: List[dict] = []

258 def tell(self, level: str, X, Y) -> "MultiFidelityActiveLearner": 

259 """Add evaluations at ``level``; charges ``costs[level]`` per row.""" 

260 X = np.atleast_2d(np.asarray(X, dtype=np.float64)) 

261 Y = _as_2d(Y) 

262 if level not in self.costs: 

263 raise ValueError(f"Unknown fidelity level {level!r}.") 

264 self.spent_[level] += self.costs[level] * Y.shape[0] 

265 if level in self.data_: 

266 Xo, Yo = self.data_[level] 

267 X, Y = np.vstack([Xo, X]), np.vstack([Yo, Y]) 

268 self.data_[level] = [X, Y] 

269 self.model_ = None 

270 return self

272 def fit(self): 

273 if self.model_ is None: 

274 self.model_ = self.model_factory().fit({k: tuple(v) for k, v in self.data_.items()}) 

275 return self.model_

277 def initial_design(self, n_per_level: Dict[str, int]) -> Dict[str, np.ndarray]: 

278 """Nested Sobol designs: each level's points are a prefix of the level below's.""" 

279 n_max = max(n_per_level.values()) 

280 X = self._space.from_unit(space_filling(n_max, self._space.n_features, "sobol", self._space._seed())) 

281 return {lvl: X[: n_per_level[lvl]] for lvl in self.levels}

283 def score(self, X) -> np.ndarray: 

284 """Acquisition for every ``(x, level)``, shape (n, n_levels).""" 

285 model = self.fit() 

286 X_hi, Y_hi = self.data_[self.top] 

287 trust = None if self.trust_z is None else level_trust(model, self.levels, X_hi, Y_hi, self.trust_z) 

288 info = fidelity_info(model, X, self.levels, self.nested, trust) 

289 if self.acquisition == "variance": 

290 return variance_reduction_per_cost(info, self.eff_costs, output_scale(self.data_[self.top][1]), self.how, self.weights) 

291 g, _ = _scalarize(self.data_[self.top][1], np.zeros_like(self.data_[self.top][1]), self.weights) 

292 best = float(g.min() if self.minimize else g.max()) 

293 return mf_expected_improvement(info, self.eff_costs, best, self.xi, self.weights, self.minimize)

295 def ask(self, remaining: Optional[float] = None): 

296 """Next query as ``(level, x)`` with ``x`` of shape (1, n_features); None if unaffordable.""" 

297 X = self._space.from_unit(space_filling(self.n_candidates, self._space.n_features, "sobol", self._space._seed())) 

298 S = self.score(X) 

299 eff = self.eff_costs 

300 affordable = np.array([remaining is None or eff[lvl] <= remaining + 1e-12 for lvl in self.levels]) 

301 if not affordable.any(): 

302 return None 

303 spent = self.total_spent 

304 if self.min_hi_fraction > 0 and affordable[-1] and spent > 0 and self.spent_[self.top] / spent < self.min_hi_fraction: 

305 affordable[:-1] = False 

306 S = np.where(affordable[None, :], S, -np.inf) 

307 i, j = np.unravel_index(int(np.argmax(S)), S.shape) 

308 self._last_scores = S 

309 return self.levels[j], X[i : i + 1]

311 def _record(self, X_test, Y_test, **extra) -> dict: 

312 rec = { 

313 "cost": self.total_spent, 

314 **{f"n_{lvl}": len(self.data_.get(lvl, [[]])[0]) for lvl in self.levels}, 

315 **{f"cost_{lvl}": self.spent_[lvl] for lvl in self.levels}, 

316 **extra, 

317 } 

318 if X_test is not None: 

319 rec.update(heldout_metrics(self.fit(), X_test, Y_test)) 

320 self.history_.append(rec) 

321 return rec

323 def run( 

324 self, 

325 simulators: Dict[str, Callable[[np.ndarray], np.ndarray]], 

326 budget: float, 

327 n_init: Optional[Dict[str, int]] = None, 

328 X_test=None, 

329 Y_test=None, 

330 callback: Optional[Callable] = None, 

331 ) -> List[dict]: 

332 """Spend ``budget`` (including the initial design) and return :attr:`history_`. 

333 

334 ``X_test`` / ``Y_test`` are high-fidelity hold-out data. 

335 """ 

336 if not self.data_: 

337 n_init = n_init or {lvl: 2 * self._space.n_features + 2 for lvl in self.levels} 

338 for lvl, X0 in self.initial_design(n_init).items(): 

339 self.tell(lvl, X0, simulators[lvl](X0)) 

340 if self.total_spent > budget: 

341 raise ValueError(f"The initial design costs {self.total_spent}, over the budget of {budget}.") 

342 rec = self._record(X_test, Y_test, level=None) 

343 if callback is not None: 

344 callback(self, rec) 

345 while True: 

346 query = self.ask(budget - self.total_spent) 

347 if query is None: 

348 break 

349 level, x = query 

350 run_levels = self.levels[: self.levels.index(level) + 1] if self.nested else [level] 

351 for lvl in run_levels: 

352 self.tell(lvl, x, simulators[lvl](x)) 

353 rec = self._record(X_test, Y_test, level=level, x=x.ravel()) 

354 if callback is not None: 

355 callback(self, rec) 

356 return self.history_
