13class BaseConformalPredictor(ABC): 

14 """Abstract base for conformal prediction wrappers. 

15 

16 All conformal predictors follow a two-phase protocol: 

17 1. fit(X, y) -- fits the underlying model AND calibrates nonconformity scores. 

18 2. predict_interval(X, alpha) -- returns (y_lower, y_upper) with 

19 guaranteed marginal coverage >= 1-alpha. 

20 

21 For multi-output regression, conformal calibration is applied independently 

22 per output (marginal coverage guarantee per target). 

23 """

60 """Point predictions. If alpha is given, returns (y_pred, y_lower, y_upper).""" 

61 if not hasattr(self, "model_"): 

62 raise AttributeError("Predictor not fitted. Call fit() first.") 

63 y_pred = self.model_.predict(X) 

64 if alpha is not None:

66 return y_pred, y_lower, y_upper 

67 return y_pred

83 @staticmethod 

84 def _conformal_quantile(scores: np.ndarray, alpha: float) -> float: 

85 """Compute the conformal quantile with finite-sample correction. 

86 

87 Returns the k-th smallest score with k = ceil((n+1)(1-alpha)), which 

88 guarantees P(Y in interval) >= 1-alpha for exchangeable data. When 

89 k > n there are too few calibration points for that guarantee, and the 

90 only valid interval is unbounded, so ``inf`` is returned. 

91 """ 

92 BaseConformalPredictor._check_alpha(alpha) 

93 scores = np.asarray(scores, dtype=float) 

94 n = len(scores) 

95 k = int(np.ceil((n + 1) * (1 - alpha))) 

96 if k > n: 

97 n_required = int(np.ceil(1 / alpha)) - 1 

98 warnings.warn( 

99 f"{n} calibration points are too few for alpha={alpha}: at least " 

100 f"{n_required} are needed for finite conformal intervals. " 

101 "Returning an infinite interval.", 

102 UserWarning, 

103 stacklevel=2, 

104 ) 

105 return float("inf") 

106 return float(np.sort(scores)[k - 1])
