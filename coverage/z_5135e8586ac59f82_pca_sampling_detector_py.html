3"""PCA view of a design of experiments, labelled with its inferred sampling method. 

4 

5PCA is linear and deterministic, so unlike UMAP it keeps relative distances and 

6needs no extra dependency, which makes it a reasonable default picture of a 

7design. The method label still comes from 

8:func:`multioutreg.sampling.detect.detect_sampling_method`: thresholds on the 

92-D embedding (nearest-neighbour spread, silhouette, entropy) depend on ``n`` 

10and ``d`` and did not separate the designs reliably, so those numbers are 

11reported as diagnostics only. 

12""" 

13

28def pca_embedding(X: np.ndarray, random_state: int = 0) -> Tuple[np.ndarray, np.ndarray]: 

29 """Standardise ``X`` and project it onto (at most) two principal components. 

30 

31 Returns ``(X_emb, explained_variance_ratio)``. ``X_emb`` always has two 

32 columns; a single-input design is padded with zeros. Constant data embeds 

33 at the origin with zero explained variance. 

34 """ 

35 X = np.asarray(X, dtype=float) 

36 X_std = StandardScaler().fit_transform(X) 

37 n_comp = min(2, X.shape[1], X.shape[0]) 

38 if not np.any(X_std): 

39 return np.zeros((X.shape[0], 2)), np.zeros(n_comp) 

40 pca = PCA(n_components=n_comp, random_state=random_state).fit(X_std) 

41 X_emb = pca.transform(X_std) 

42 if n_comp < 2: 

43 X_emb = np.column_stack([X_emb, np.zeros((X.shape[0], 2 - n_comp))]) 

44 return X_emb, pca.explained_variance_ratio_

47def embedding_diagnostics(X_emb: np.ndarray, random_state: int = 0) -> Dict[str, float]: 

48 """Structure of a 2-D embedding: nearest-neighbour spread, 2-means silhouette, entropy. 

49 

50 Descriptive only; see the module docstring for why they are not used to 

51 classify the design. 

52 """ 

53 n_distinct = np.unique(X_emb, axis=0).shape[0] 

54 dists = KDTree(X_emb).query(X_emb, k=2)[0][:, 1] 

55 sil = 0.0 

56 if n_distinct >= 3: 

57 clusters = KMeans(n_clusters=2, n_init=10, random_state=random_state).fit_predict(X_emb) 

58 if len(np.unique(clusters)) > 1: 

59 sil = float(silhouette_score(X_emb, clusters)) 

60 return { 

61 "std": float(np.std(dists)), 

62 "silhouette": sil, 

63 "entropy": float(entropy(np.histogram(dists, bins=30, density=True)[0])), 

64 }

67def generate_sampling_plot_and_metrics( 

68 X: np.ndarray, 

69 random_state: int = 42, 

70 labels: Optional[np.ndarray] = None, 

71) -> Tuple[plt.Figure, dict]: 

72 """Plot the PCA projection of a design and infer its sampling method. 

73 

74 Parameters 

75 ---------- 

76 X : np.ndarray 

77 Input (design) matrix. Pass only input columns, not responses. 

78 random_state : int 

79 Seed for PCA, the diagnostics and the simulated reference designs. 

80 labels : array-like or None 

81 Optional per-point values to colour the points by. 

82 

83 Returns 

84 ------- 

85 Tuple[plt.Figure, dict] 

86 The figure and a dict with ``method``, ``explanation``, ``projection`` 

87 (``"PCA"``), ``explained_variance``, ``p_values`` and the embedding 

88 diagnostics ``std``, ``silhouette`` and ``entropy``. 

89 """ 

90 if X is None or len(X) == 0: 

91 raise ValueError("generate_sampling_plot_and_metrics needs a non-empty input matrix") 

92 X = np.asarray(X, dtype=float) 

93 if X.ndim == 1: 

94 X = X[:, None] 

95 

96 try: 

97 detection = detect_sampling_method(X, random_state=random_state) 

98 method, explanation, p_values = detection.method, detection.explanation, detection.p_values 

99 except ValueError as exc: # e.g. fewer than 4 points 

100 method, explanation, p_values = "Uncertain", f"Sampling method not tested: {exc} -> Uncertain", {} 

101 

102 X_emb, evr = pca_embedding(X, random_state=random_state) 

103 

104 fig, ax = plt.subplots(figsize=(6, 5)) 

105 if labels is None: 

106 ax.scatter(X_emb[:, 0], X_emb[:, 1], alpha=0.7, edgecolor="k", linewidth=0.3) 

107 else: 

108 sc = ax.scatter(X_emb[:, 0], X_emb[:, 1], c=label_colour_values(labels), cmap="viridis", 

109 alpha=0.7, edgecolor="k", linewidth=0.3) 

110 fig.colorbar(sc, ax=ax) 

111 ax.set_title(f"PCA Projection - Inferred: {method}") 

112 ax.set_xlabel(f"PC1 ({evr[0]:.0%} var)" if len(evr) > 0 else "PC1") 

113 ax.set_ylabel(f"PC2 ({evr[1]:.0%} var)" if len(evr) > 1 else "") 

114 fig.tight_layout() 

115 

116 return fig, { 

117 "method": method, 

118 "explanation": explanation, 

119 "projection": "PCA", 

120 "explained_variance": [float(v) for v in evr], 

121 "p_values": {k: float(v) for k, v in p_values.items()}, 

122 **embedding_diagnostics(X_emb, random_state=random_state), 

123 }

126def label_colour_values(labels) -> np.ndarray: 

127 """Numeric colour values for ``labels`` (categories become their codes).""" 

128 labels = np.asarray(labels) 

129 if np.issubdtype(labels.dtype, np.number): 

130 return labels.astype(float) 

131 return np.unique(labels.astype(str), return_inverse=True)[1]
