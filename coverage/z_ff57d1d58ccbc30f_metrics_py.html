3import numpy as np 

4import pandas as pd 

5from typing import Any 

6import uncertainty_toolbox as uct 

7 

8def get_uq_performance_metrics( 

9 model: Any, 

10 X_test: np.ndarray, 

11 y_test: np.ndarray, 

12 std_method: str ='return_std', 

13): 

14 """ 

15 Computes uncertainty-toolbox performance metrics for each output of a multioutput regressor. 

16 

17 Args: 

18 model: MultiOutputRegressor or similar with predict(X, return_std=...) support 

19 X_test: Test features, shape (n_samples, n_features) 

20 y_test: Test targets, shape (n_samples, n_outputs) 

21 std_method: Which kwarg to use for uncertainty (default 'return_std') 

22 

23 Returns: 

24 metrics_df: pd.DataFrame with metrics per output 

25 metrics_overall: dict of metrics computed on all outputs at once (flattened) 

26 """ 

27 # Predict means and stds 

28 try: 

29 # Some regressors require return_std=True, others return_uncertainty=True 

30 y_pred, y_std = model.predict(X_test, **{std_method: True}) 

31 except TypeError: 

32 # Try 'return_std' as fallback if user passed wrong std_method 

33 y_pred, y_std = model.predict(X_test, return_std=True) 

34 

35 y_true = np.asarray(y_test) 

36 y_pred = np.asarray(y_pred) 

37 y_std = np.asarray(y_std) 

38 n_outputs = y_true.shape[1] 

39 

40 metrics_list = [] 

41 for i in range(n_outputs): 

42 metrics = uct.get_all_metrics( 

43 y_pred[:, i], 

44 y_std[:, i], 

45 y_true[:, i] 

46 ) 

47 metrics['output'] = i 

48 metrics_list.append(metrics) 

49 

50 metrics_df = pd.DataFrame(metrics_list) 

51 

52 # Overall metrics on flattened arrays 

53 overall_metrics = uct.get_all_metrics( 

54 y_pred.flatten(), 

55 y_std.flatten(), 

56 y_true.flatten() 

57 ) 

58 

59 print(metrics_df) 

60 print("Overall metrics:", overall_metrics) 

61 return metrics_df, overall_metrics 

62
