28class ProphetForecaster: 

29 """Thin sklearn-style wrapper around Meta Prophet. 

30 

31 Prophet handles trend and seasonality automatically via its additive 

32 decomposition model. Quantile uncertainty is obtained from Prophet's 

33 built-in prediction interval sampling. 

34 

35 Requires:: 

36 

37 pip install prophet 

38 

39 Parameters 

40 ---------- 

41 seasonality_mode : str, default "additive" 

42 ``"additive"`` or ``"multiplicative"``. 

43 interval_width : float, default 0.8 

44 Width of the inner prediction interval that Prophet uses internally 

45 (``yhat_lower`` / ``yhat_upper``). Wider = larger fan. 

46 mcmc_samples : int, default 0 

47 If 0, uses MAP estimation (fast). Set > 0 for full Bayesian 

48 posterior (slow but better-calibrated intervals). 

49 additional_seasonalities : list[dict] | None 

50 Extra seasonality components passed to ``Prophet.add_seasonality()``. 

51 

52 Examples 

53 -------- 

54 >>> import pandas as pd, numpy as np 

55 >>> dates = pd.date_range("2022-01-01", periods=200, freq="D") 

56 >>> y = pd.Series(np.sin(np.linspace(0, 4*np.pi, 200)), index=dates) 

57 >>> f = ProphetForecaster().fit(y) 

58 >>> result = f.predict(prediction_length=30) 

59 >>> result.quantiles.shape # (1, 3, 30) 

60 """

79 def fit( 

80 self, 

81 y: pd.Series | np.ndarray, 

82 datetime_index: Optional[pd.DatetimeIndex] = None, 

83 freq: str = "D", 

84 ) -> "ProphetForecaster": 

85 """Fit Prophet model. 

86 

87 Parameters 

88 ---------- 

89 y : 1D array or pd.Series 

90 If pd.Series with DatetimeIndex, the index is used automatically. 

91 If np.ndarray, ``datetime_index`` must be provided or a daily 

92 range starting from today is synthesised. 

93 datetime_index : pd.DatetimeIndex | None 

94 Explicit datetime index (used when ``y`` is a plain array). 

95 freq : str, default "D" 

96 Frequency string used when synthesising a date range. 

97 

98 Returns 

99 ------- 

100 self 

101 """ 

102 _require_prophet() 

103 

104 if isinstance(y, pd.Series) and isinstance(y.index, pd.DatetimeIndex): 

105 idx = y.index 

106 vals = y.to_numpy(dtype=float) 

107 self._freq = pd.infer_freq(idx) or freq 

108 elif datetime_index is not None: 

109 idx = datetime_index 

110 vals = np.asarray(y, dtype=float) 

111 self._freq = pd.infer_freq(idx) or freq 

112 else: 

113 n = len(y) 

114 vals = np.asarray(y, dtype=float) 

115 idx = pd.date_range("2000-01-01", periods=n, freq=freq) 

116 self._freq = freq 

117 

118 df_prophet = pd.DataFrame({"ds": idx, "y": vals}) 

119 self._last_date = df_prophet["ds"].iloc[-1] 

120 

121 self._model = _Prophet( 

122 seasonality_mode=self.seasonality_mode, 

123 interval_width=self.interval_width, 

124 mcmc_samples=self.mcmc_samples, 

125 ) 

126 for s in self.additional_seasonalities: 

127 self._model.add_seasonality(**s) 

128 

129 import logging 

130 # Suppress Stan / Prophet verbose output 

131 logging.getLogger("prophet").setLevel(logging.WARNING) 

132 logging.getLogger("cmdstanpy").setLevel(logging.WARNING) 

133 self._model.fit(df_prophet) 

134 return self

136 def predict( 

137 self, 

138 prediction_length: int, 

139 quantiles: Sequence[float] = (0.1, 0.5, 0.9), 

140 ) -> ForecastResult: 

141 """Forecast ``prediction_length`` steps ahead. 

142 

143 Prophet's ``yhat_lower`` / ``yhat_upper`` are mapped to the outermost 

144 requested quantiles; ``yhat`` becomes the median. 

145 

146 Parameters 

147 ---------- 

148 prediction_length : int 

149 quantiles : sequence of float 

150 

151 Returns 

152 ------- 

153 ForecastResult, shape ``[1, len(quantiles), prediction_length]`` 

154 """ 

155 if self._model is None: 

156 raise RuntimeError("Call fit() before predict().") 

157 _require_prophet() 

158 

159 future = self._model.make_future_dataframe( 

160 periods=prediction_length, freq=self._freq or "D", include_history=False 

161 ) 

162 forecast = self._model.predict(future) 

163 

164 yhat = forecast["yhat"].to_numpy()[:prediction_length] 

165 lower = forecast["yhat_lower"].to_numpy()[:prediction_length] 

166 upper = forecast["yhat_upper"].to_numpy()[:prediction_length] 

167 

168 q_levels = tuple(float(q) for q in quantiles) 

169 q_arr = _map_quantiles(yhat, lower, upper, q_levels) 

170 

171 return ForecastResult( 

172 quantiles=q_arr[np.newaxis, :, :], # [1, Q, H] 

173 q_levels=q_levels, 

174 ids=("y",), 

175 )

178def _map_quantiles( 

179 median: np.ndarray, 

180 lower: np.ndarray, 

181 upper: np.ndarray, 

182 q_levels: tuple, 

183) -> np.ndarray: 

184 """Map (lower, median, upper) to arbitrary quantile levels by linear interpolation. 

185 

186 Treats lower=q=0.1, median=q=0.5, upper=q=0.9 as anchor points and 

187 linearly interpolates for other levels. 

188 """ 

189 anchors = np.array([0.1, 0.5, 0.9]) 

190 anchor_vals = np.stack([lower, median, upper], axis=0) # (3, H) 

191 

192 q_arr = np.zeros((len(q_levels), len(median))) 

193 for i, q in enumerate(q_levels): 

194 q_arr[i] = np.array([ 

195 np.interp(q, anchors, anchor_vals[:, h]) 

196 for h in range(len(median)) 

197 ]) 

198 return q_arr
