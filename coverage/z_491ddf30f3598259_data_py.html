28 

29 ``run`` holds metadata (title, model, data hash, seed, version, date, CV

59 def to_dict(self, include_images: bool = True) -> Dict[str, Any]: 

60 """Plain-Python copy: numpy converted, NaN/inf as ``None``, floats in 

61 arrays rounded to 4 significant figures.""" 

62 d = _clean(asdict(self)) 

63 if not include_images: 

64 d["figures"] = [{k: v for k, v in f.items() if k != "img_b64"} for f in d["figures"]] 

65 if d.get("pca"): 

66 d["pca"] = {k: v for k, v in d["pca"].items() if k != "plot_b64"} 

67 if d.get("sampling"): 

68 d["sampling"] = {k: v for k, v in d["sampling"].items() if k != "plots"} 

69 return d

85def _round(x: float) -> Optional[float]: 

86 if not math.isfinite(x): 

87 return None 

88 return float(f"{x:.4g}")

91def _clean(obj: Any, in_array: bool = False) -> Any: 

92 """Make ``obj`` JSON-safe. Floats inside lists are rounded (they are chart 

93 data); scalar floats keep full precision.""" 

94 if isinstance(obj, dict): 

95 return {str(k): _clean(v) for k, v in obj.items()} 

96 if isinstance(obj, (list, tuple)): 

97 return [_clean(v, in_array=True) for v in obj] 

98 if isinstance(obj, np.ndarray): 

99 return [_clean(v, in_array=True) for v in obj.tolist()] 

100 if isinstance(obj, (bool, np.bool_)): 

101 return bool(obj) 

102 if isinstance(obj, (int, np.integer)): 

103 return int(obj) 

104 if isinstance(obj, (float, np.floating)): 

105 x = float(obj) 

106 if not math.isfinite(x): 

107 return None 

108 return _round(x) if in_array else x 

109 return obj

112def downsample_index(n: int, cap: int = MAX_POINTS, seed: int = 0) -> np.ndarray: 

113 """Sorted indices of at most ``cap`` of ``n`` points (all of them if n <= cap).""" 

114 if n <= cap: 

115 return np.arange(n) 

116 return np.sort(np.random.default_rng(seed).choice(n, cap, replace=False))
