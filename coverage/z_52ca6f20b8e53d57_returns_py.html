2"""Price <-> change transforms for forecasting financial series. 

3 

4Prices are close to a random walk: non-stationary, with a scale that drifts. 

5Models are fitted to the changes instead -- simple differences or log returns 

6-- and their forecast is cumulated back onto the last observed level: 

7 

8 log returns r_t = log(p_t / p_{t-1}) -> p_{T+k} = p_T * exp(r_{T+1} + ... + r_{T+k}) 

9 differences d_t = p_t - p_{t-1} -> p_{T+k} = p_T + d_{T+1} + ... + d_{T+k} 

10""" 

11 

12from __future__ import annotations 

13 

14import numpy as np 

15import pandas as pd 

16 

17#: Transforms accepted by :func:`to_changes` / :func:`to_levels`. 

18TRANSFORMS = ("none", "diff", "log-return") 

19 

20 

21def _check(kind: str) -> None: 

22 if kind not in TRANSFORMS: 

23 raise ValueError(f"transform must be one of {TRANSFORMS}, got {kind!r}") 

24 

25 

26def to_changes(levels: pd.Series, kind: str) -> pd.Series: 

27 """Period-to-period changes of *levels*; the first period is dropped. 

28 

29 ``kind='none'`` returns *levels* unchanged. Log returns need every level 

30 to be positive. 

31 """ 

32 _check(kind) 

33 if kind == "none": 

34 return levels 

35 if kind == "log-return": 

36 if (levels <= 0).any(): 

37 raise ValueError("log returns need strictly positive values; use --transform diff") 

38 return np.log(levels).diff().iloc[1:] 

39 return levels.diff().iloc[1:] 

40 

41 

42def to_levels(last_level: float, changes, kind: str) -> np.ndarray: 

43 """Cumulate forecast *changes* onto *last_level*: the forecast as levels.""" 

44 _check(kind) 

45 changes = np.asarray(changes, dtype=float) 

46 if kind == "none": 

47 return changes 

48 if kind == "log-return": 

49 return float(last_level) * np.exp(np.cumsum(changes)) 

50 return float(last_level) + np.cumsum(changes)
