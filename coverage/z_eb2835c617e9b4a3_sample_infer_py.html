33def infer_sampling(df, num_cols): 

34 # Grid: unique value count per feature matches grid size, mesh pattern in plots 

35 n_unique_vals = [len(df[col].unique()) for col in num_cols] 

36 grid_like = np.prod(n_unique_vals) == len(df) 

37 # LHS: unique values per feature, each only used once; uniform histogram 

38 lhs_like = all(df[col].value_counts().max() == 1 for col in num_cols) 

39 # For 'other', neither pattern 

40 if grid_like: 

41 st.info("Sampling looks like grid sampling (regular mesh in value pairs, distinct fixed values per feature).") 

42 elif lhs_like: 

43 st.info("Sampling looks like Latin hypercube sampling (one value per bin per feature, uniform marginals).") 

44 else: 

45 st.info("Sampling does not match grid or LHS. It may be a random or other custom scheme.")
