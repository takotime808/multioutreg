25class TorchStandardScaler: 

26 """StandardScaler for numpy arrays used before/after PyTorch forward passes.""" 

27 

28 def fit(self, X: np.ndarray) -> "TorchStandardScaler": 

29 self.mean_ = X.mean(axis=0) 

30 self.scale_ = X.std(axis=0) 

31 self.scale_[self.scale_ == 0] = 1.0 

32 return self 

33 

34 def transform(self, X: np.ndarray) -> np.ndarray: 

35 return (X - self.mean_) / self.scale_ 

36 

37 def inverse_transform(self, X: np.ndarray) -> np.ndarray: 

38 return X * self.scale_ + self.mean_ 

39 

40 def fit_transform(self, X: np.ndarray) -> np.ndarray: 

41 return self.fit(X).transform(X)

44def train_loop( 

45 model: "nn.Module", 

46 optimizer: "torch.optim.Optimizer", 

47 loss_fn, 

48 X_tensor: "torch.Tensor", 

49 Y_tensor: "torch.Tensor", 

50 max_epochs: int = 500, 

51 batch_size: int = 64, 

52 patience: int = 20, 

53 min_delta: float = 1e-6, 

54 lr_scheduler: "torch.optim.lr_scheduler._LRScheduler | None" = None, 

55) -> list[float]: 

56 """Standard mini-batch training loop with early stopping. 

57 

58 Parameters 

59 ---------- 

60 model : nn.Module 

61 optimizer : torch.optim.Optimizer 

62 loss_fn : callable 

63 Called as ``loss_fn(y_pred, y_true)`` and returns a scalar tensor. 

64 X_tensor, Y_tensor : torch.Tensor 

65 Full training data (will be batched internally). 

66 max_epochs : int 

67 batch_size : int 

68 patience : int 

69 Number of epochs without improvement before stopping. 

70 min_delta : float 

71 Minimum absolute improvement to reset patience counter. 

72 lr_scheduler : optional 

73 Called as ``lr_scheduler.step()`` after each epoch. 

74 

75 Returns 

76 ------- 

77 losses : list[float] 

78 Training loss per epoch. 

79 """ 

80 require_torch() 

81 n = X_tensor.shape[0] 

82 losses = [] 

83 best_loss = float("inf") 

84 epochs_no_improve = 0 

85 

86 for epoch in range(max_epochs): 

87 model.train() 

88 perm = torch.randperm(n) 

89 epoch_loss = 0.0 

90 n_batches = 0 

91 for start in range(0, n, batch_size): 

92 idx = perm[start : start + batch_size] 

93 X_batch = X_tensor[idx] 

94 Y_batch = Y_tensor[idx] 

95 optimizer.zero_grad() 

96 y_pred = model(X_batch) 

97 loss = loss_fn(y_pred, Y_batch) 

98 loss.backward() 

99 optimizer.step() 

100 epoch_loss += loss.item() 

101 n_batches += 1 

102 

103 epoch_loss /= max(n_batches, 1) 

104 losses.append(epoch_loss) 

105 

106 if lr_scheduler is not None: 

107 lr_scheduler.step() 

108 

109 if best_loss - epoch_loss > min_delta: 

110 best_loss = epoch_loss 

111 epochs_no_improve = 0 

112 else: 

113 epochs_no_improve += 1 

114 if epochs_no_improve >= patience: 

115 break 

116 

117 return losses

131def split_mean_var(out: "torch.Tensor", n_outputs: int): 

132 """Split a ``(n, 2 * n_outputs)`` network output into mean and variance. 

133 

134 The second half is passed through softplus (plus a small floor), so the 

135 variance is always positive. 

136 """ 

137 mean = out[:, :n_outputs] 

138 var = torch.nn.functional.softplus(out[:, n_outputs:]) + _MIN_VAR 

139 return mean, var

142def gaussian_nll_loss(n_outputs: int): 

143 """Loss for a mean/variance head: Gaussian negative log-likelihood. 

144 

145 Training on the NLL lets the network learn the noise variance (aleatoric 

146 uncertainty) as a function of x (Nix & Weigend 1994; Lakshminarayanan 

147 et al. 2017). The returned callable has the ``loss_fn(y_pred, y_true)`` 

148 signature expected by :func:`train_loop`. 

149 """ 

150 require_torch() 

151 nll = nn.GaussianNLLLoss(eps=_MIN_VAR) 

152 

153 def loss_fn(out, y): 

154 mean, var = split_mean_var(out, n_outputs) 

155 return nll(mean, y, var) 

156 

157 return loss_fn

160def mixture_mean_std(means: np.ndarray, variances: np.ndarray): 

161 """Moments of an equal-weight Gaussian mixture over axis 0. 

162 

163 ``means`` and ``variances`` have shape ``(n_members, n_samples, n_outputs)``. 

164 Total variance = mean noise variance (aleatoric) + variance of the member 

165 means (epistemic). 

166 """ 

167 mean = means.mean(axis=0) 

168 var = variances.mean(axis=0) + means.var(axis=0) 

169 return mean, np.sqrt(var)
