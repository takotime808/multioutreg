3import numpy as np 

4import matplotlib.pyplot as plt 

5from sklearn.decomposition import PCA 

6from multioutreg.utils.figure_utils import plot_to_b64 

7 

8def generate_pca_variance_plot( 

9 pca: PCA, 

10 n_selected: int | None = None, 

11 threshold: float | None = None, 

12) -> str: 

13 """ 

14 Generate a base64-encoded bar plot showing the explained variance ratio of each principal component. 

15 

16 Parameters 

17 ---------- 

18 pca : PCA 

19 Fitted PCA instance. 

20 n_selected : int | None, optional 

21 Number of components retained. If provided, a vertical dashed line is 

22 drawn on the scree plot. 

23 threshold : float | None, optional 

24 Explained variance threshold. When given, a horizontal dashed line is 

25 drawn to illustrate the cutoff. 

26 

27 Returns 

28 ------- 

29 str 

30 Base64 encoded PNG image of the scree plot. 

31 """ 

32 def plot_fn(): 

33 comps = np.arange(1, len(pca.explained_variance_ratio_) + 1) 

34 height = max(4, 0.3 * len(comps)) # Dynamic height 

35 plt.figure(figsize=(int(1.5*len(comps)), height)) 

36 plt.plot(comps, pca.explained_variance_ratio_, marker="o") 

37 plt.xlabel("Principal Component", fontsize=10) 

38 plt.ylabel("Explained Variance Ratio", fontsize=10) 

39 plt.yticks(comps, [f"PC{i}" for i in comps]) # Smaller y-axis font 

40 plt.title("PCA Scree Plot") 

41 if n_selected is not None: 

42 plt.axvline(n_selected, color="red", linestyle="--", label=f"k = {n_selected}") 

43 if threshold is not None: 

44 plt.axhline(threshold, color="green", linestyle=":", label=f"threshold = {threshold:.2f}") 

45 if n_selected is not None or threshold is not None: 

46 plt.legend() 

47 plt.tight_layout() 

48 

49 return plot_to_b64(plot_fn)
