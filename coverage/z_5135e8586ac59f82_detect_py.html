3"""Infer how a design of experiments was sampled (grid, LHS, Sobol, random). 

4 

5The observed design is summarised by three statistics, and each is compared 

6with the same statistics computed on designs simulated by every candidate 

7method with the same ``n`` and ``d`` (a parametric bootstrap / Monte Carlo 

8null). This replaces ad-hoc absolute thresholds, which cannot be valid for 

9every ``n`` and ``d``. 

10 

11Statistics (on inputs mapped to the unit cube): 

12 

13- ``strata``: mean over dimensions of the fraction of the ``n`` equal-width 

14 strata that contain a point. LHS fills all of them; i.i.d. uniform fills 

15 about ``1 - 1/e`` (0.63). 

16- ``log_cd``: log10 of the centered L2 discrepancy (``scipy.stats.qmc``), which 

17 measures joint uniformity. Sobol < LHS < random. Skipped for large ``n`` 

18 (cost is O(n^2 d)). 

19- ``nn``: mean nearest-neighbour distance (from a random subset of query 

20 points when ``n`` is large). Space-filling designs spread their 

21 points out more than random ones do. 

22 

23Candidates: ``Random`` (i.i.d. uniform), ``LHS`` and ``Sobol``, where "Sobol" 

24means the low-discrepancy family (the null mixes scrambled Sobol and Halton). 

25Unscrambled Sobol / Halton sequences are deterministic and are matched 

26exactly. Grids are recognised structurally (few distinct levels per input). 

27 

28For each candidate the statistics are combined into one Mahalanobis distance 

29from the simulated designs' mean (using their covariance), and the Monte Carlo 

30p-value is the fraction of simulated designs at least as far away. A method is 

31"consistent" with the data when its p-value is at least ``alpha``; the reported method is the one with the largest p-value, 

32or ``"Uncertain"`` when no candidate is consistent (e.g. observational data or 

33non-uniform marginals). 

34 

35Limitations: the tests assume the design was drawn on a box with uniform 

36marginals. When bounds are not given they are estimated from the data's 

37min/max (the simulated designs go through the same transform, so the null is 

38matched). Designs with other marginals must be mapped to the unit cube first, 

39e.g. via their CDFs (see ``mor_cli/use_ks_for_sample_bias.py``). 

40"""

53CANDIDATES: Tuple[str, ...] = ("Random", "LHS", "Sobol") 

54# Centered discrepancy is O(n^2 d); above this many points it is skipped. 

55MAX_N_DISCREPANCY = 1000 

56# Below this many points the Monte Carlo tests have almost no power. 

57MIN_POINTS_MC = 10 

58# Column names treated as responses (not part of the design) by guess_input_columns. 

59RESPONSE_NAME = re.compile(r"^(y\d*|y_.*|response.*|output.*|target.*|label)$", re.IGNORECASE) 

60# Mean nearest-neighbour distance is estimated from at most this many query points. 

61MAX_NN_QUERIES = 500

76def guess_input_columns(columns: Sequence[str]) -> list: 

77 """Drop columns whose names look like responses (``y``, ``y1``, ``response``, ``target``...). 

78 

79 Falls back to all columns when every name looks like a response. 

80 """ 

81 inputs = [c for c in columns if not RESPONSE_NAME.match(str(c))] 

82 return inputs or list(columns)

85def to_unit_cube(X: np.ndarray, bounds: Optional[Tuple[np.ndarray, np.ndarray]] = None) -> np.ndarray: 

86 """Map ``X`` to [0, 1]^d. 

87 

88 With ``bounds=(lower, upper)`` this is a plain affine map. Without bounds the 

89 sample min/max are placed at ``0.5/n`` and ``1 - 0.5/n``, i.e. the centres of 

90 the outermost strata, which is where an LHS puts them on average. 

91 """ 

92 X = np.asarray(X, dtype=float) 

93 if bounds is not None: 

94 lo, hi = (np.broadcast_to(np.asarray(b, dtype=float), X.shape[1:]) for b in bounds) 

95 if np.any(hi <= lo): 

96 raise ValueError("Every upper bound must exceed its lower bound.") 

97 return np.clip((X - lo) / (hi - lo), 0.0, 1.0) 

98 n = X.shape[0] 

99 lo, hi = X.min(axis=0), X.max(axis=0) 

100 span = np.where(hi > lo, hi - lo, 1.0) 

101 return (X - lo) / span * (n - 1) / n + 0.5 / n

104def design_statistics(U: np.ndarray, rng: Optional[np.random.Generator] = None) -> Dict[str, float]: 

105 """Statistics of a design already on the unit cube (see module docstring).""" 

106 n, d = U.shape 

107 bins = np.minimum((U * n).astype(int), n - 1) 

108 strata = float(np.mean([np.unique(bins[:, j]).size / n for j in range(d)])) 

109 stats = {"strata": strata} 

110 if n <= MAX_N_DISCREPANCY: 

111 stats["log_cd"] = float(np.log10(max(qmc.discrepancy(U, method="CD"), 1e-300))) 

112 queries = U 

113 if n > MAX_NN_QUERIES: 

114 rng = rng if rng is not None else np.random.default_rng(0) 

115 queries = U[rng.choice(n, MAX_NN_QUERIES, replace=False)] 

116 nn = cKDTree(U).query(queries, k=2)[0][:, 1] 

117 stats["nn"] = float(np.mean(nn)) 

118 return stats

121def _simulate(method: str, n: int, d: int, rng: np.random.Generator) -> np.ndarray: 

122 if method == "Random": 

123 return rng.random((n, d)) 

124 if method == "LHS": 

125 return qmc.LatinHypercube(d=d, seed=rng).random(n) 

126 if method == "Sobol": 

127 # "Sobol" stands for the low-discrepancy (QMC) family: half the null 

128 # designs are scrambled Sobol, half scrambled Halton. 

129 if rng.random() < 0.5: 

130 return qmc.Halton(d=d, seed=rng).random(n) 

131 with warnings.catch_warnings(): 

132 # n not a power of 2 loses balance properties; that is what we want to mimic. 

133 warnings.simplefilter("ignore", UserWarning) 

134 return qmc.Sobol(d=d, scramble=True, seed=rng).random(n) 

135 raise ValueError(f"Unknown sampling method: {method}")

138def default_n_sim(n: int) -> int: 

139 """Number of simulated designs per candidate, fewer for large ``n`` to bound runtime. 

140 

141 The statistics are more concentrated for large ``n``, and 49 simulations 

142 still give a smallest attainable p-value of 0.02. 

143 """ 

144 if n <= 500: 

145 return 199 

146 if n <= 2000: 

147 return 99 

148 return 49

151def _mc_p_value(observed: np.ndarray, simulated: np.ndarray) -> float: 

152 """Monte Carlo p-value of a statistic vector against simulated ones. 

153 

154 Distance is Mahalanobis from the simulated mean; the covariance gets a small 

155 ridge because some statistics are (nearly) constant under a method, e.g. an 

156 LHS on known bounds always fills every stratum. Uses the (count + 1) / (B + 1) 

157 correction, so the smallest attainable p-value is 1 / (B + 1). 

158 """ 

159 mu = simulated.mean(axis=0) 

160 cov = np.atleast_2d(np.cov(simulated, rowvar=False)) 

161 scale = np.maximum(np.diag(cov), (1e-3 * np.abs(mu)) ** 2 + 1e-12) 

162 cov = cov + np.diag(1e-2 * scale) 

163 prec = np.linalg.inv(cov) 

164 

165 def dist(v: np.ndarray) -> np.ndarray: 

166 diff = np.atleast_2d(v) - mu 

167 return np.einsum("ij,jk,ik->i", diff, prec, diff) 

168 

169 d_obs = dist(observed)[0] 

170 d_sim = dist(simulated) 

171 return float((np.sum(d_sim >= d_obs) + 1) / (simulated.shape[0] + 1))

174def matches_unscrambled_qmc(U: np.ndarray, bounds_known: bool) -> Optional[str]: 

175 """Name of the unscrambled QMC sequence whose first ``n`` points equal ``U`` (row order ignored). 

176 

177 Unscrambled sequences are deterministic, so they cannot be represented by a 

178 Monte Carlo null; they are recognised exactly instead. 

179 """ 

180 n, d = U.shape 

181 rows = np.lexsort(U.T[::-1]) 

182 with warnings.catch_warnings(): 

183 warnings.simplefilter("ignore", UserWarning) 

184 refs = { 

185 "Sobol": qmc.Sobol(d=d, scramble=False).random(n + 1), 

186 "Halton": qmc.Halton(d=d, scramble=False).random(n + 1), 

187 } 

188 for name, ref in refs.items(): 

189 for R in (ref[:n], ref[1:]): # some tools skip the first point (the origin) 

190 if not bounds_known: 

191 R = to_unit_cube(R) 

192 R = R[np.lexsort(R.T[::-1])] 

193 if np.allclose(U[rows], R, atol=1e-6): 

194 return name 

195 return None

198def grid_structure(X: np.ndarray) -> Tuple[bool, bool, np.ndarray]: 

199 """Check for grid structure. 

200 

201 Returns ``(grid_like, full_factorial, levels)``: ``grid_like`` when every 

202 column takes at most ``n / 2`` distinct values (continuous designs have 

203 ``n``); ``full_factorial`` when the rows are exactly the Cartesian product 

204 of those levels. 

205 """ 

206 X = np.asarray(X, dtype=float) 

207 n = X.shape[0] 

208 levels = np.array([np.unique(X[:, j]).size for j in range(X.shape[1])]) 

209 grid_like = bool(np.all((levels >= 2) & (levels <= n / 2))) 

210 full = False 

211 if grid_like and int(np.prod(levels.astype(float))) == n: 

212 full = np.unique(X, axis=0).shape[0] == n 

213 return grid_like, bool(full), levels

216def detect_sampling_method( 

217 X: np.ndarray, 

218 bounds: Optional[Tuple[np.ndarray, np.ndarray]] = None, 

219 n_sim: Optional[int] = None, 

220 alpha: float = 0.05, 

221 candidates: Sequence[str] = CANDIDATES, 

222 random_state: Optional[int] = 0, 

223) -> SamplingDetection: 

224 """Infer the sampling method of a design ``X`` (rows = points, columns = inputs). 

225 

226 Pass only the *input* columns; responses are not part of the design and 

227 distort every statistic. ``bounds=(lower, upper)`` should be given when the 

228 design region is known (e.g. ``(0, 1)`` after a probability integral 

229 transform); otherwise it is estimated from the data. ``n_sim`` simulated 

230 designs are drawn per candidate (default :func:`default_n_sim`). 

231 

232 The result is a statistical statement, not proof: "consistent with LHS" 

233 means LHS-simulated designs of the same size look like this one, and with 

234 few points several methods can be consistent at once. 

235 """ 

236 if X is None: 

237 raise ValueError("X is required") 

238 X = np.asarray(X, dtype=float) 

239 if X.ndim == 1: 

240 X = X[:, None] 

241 if X.shape[0] < 4 or X.shape[1] == 0: 

242 raise ValueError(f"Need at least 4 points and 1 column, got shape {X.shape}") 

243 finite = np.all(np.isfinite(X), axis=1) 

244 if not finite.all(): 

245 X = X[finite] 

246 if X.shape[0] < 4: 

247 raise ValueError( 

248 f"Need at least 4 rows without missing or infinite values, got {X.shape[0]}" 

249 ) 

250 n = X.shape[0] 

251 

252 grid_like, full, levels = grid_structure(X) 

253 if grid_like: 

254 kind = "full-factorial grid" if full else "grid-like design (few repeated levels per input)" 

255 return SamplingDetection( 

256 method="Grid", 

257 explanation=( 

258 f"Every input takes only {levels.min()}-{levels.max()} distinct values over {n} points: " 

259 f"{kind} -> Grid" 

260 ), 

261 statistics={"levels_min": float(levels.min()), "levels_max": float(levels.max())}, 

262 ) 

263 

264 # Inputs with few repeated levels (discrete / categorical) break the 

265 # continuous statistics; test the continuous inputs only. 

266 continuous = np.where(levels > n / 2)[0] 

267 discrete = np.where(levels <= n / 2)[0] 

268 if discrete.size >= 2 and grid_structure(X[:, discrete])[1]: 

269 return SamplingDetection( 

270 method="Grid", 

271 explanation=( 

272 f"Columns {discrete.tolist()} form a full-factorial grid of {n} points; columns " 

273 f"{continuous.tolist()} vary continuously (derived features or responses?) -> Grid" 

274 ), 

275 statistics={"levels_min": float(levels[discrete].min()), "levels_max": float(levels[discrete].max())}, 

276 columns_used=discrete.tolist(), 

277 ) 

278 if continuous.size == 0: 

279 return SamplingDetection( 

280 method="Uncertain", 

281 explanation="No continuous input columns to test -> Uncertain", 

282 ) 

283 Xc = X[:, continuous] 

284 if bounds is not None: 

285 lo, hi = (np.broadcast_to(np.asarray(b, dtype=float), (X.shape[1],))[continuous] for b in bounds) 

286 cbounds = (lo, hi) 

287 else: 

288 cbounds = None 

289 d = Xc.shape[1] 

290 U = to_unit_cube(Xc, cbounds) 

291 note = "" if continuous.size == X.shape[1] else f" (tested {d} of {X.shape[1]} inputs; others are discrete)" 

292 

293 exact = matches_unscrambled_qmc(U, cbounds is not None) 

294 if exact is not None: 

295 return SamplingDetection( 

296 method="Sobol", 

297 explanation=f"Points are exactly the first {n} points of the unscrambled {exact} sequence{note} -> Sobol", 

298 statistics=design_statistics(U), 

299 columns_used=continuous.tolist(), 

300 ) 

301 

302 rng = np.random.default_rng(random_state) 

303 n_sim = default_n_sim(n) if n_sim is None else int(n_sim) 

304 if n < MIN_POINTS_MC: 

305 return SamplingDetection( 

306 method="Uncertain", 

307 explanation=f"Only {n} points: too few to tell Random, LHS and Sobol apart{note} -> Uncertain", 

308 columns_used=continuous.tolist(), 

309 ) 

310 observed = design_statistics(U, rng) 

311 

312 p_values: Dict[str, float] = {} 

313 for method in candidates: 

314 sims = [] 

315 for _ in range(n_sim): 

316 S = _simulate(method, n, d, rng) 

317 if cbounds is None: 

318 S = to_unit_cube(S) 

319 sims.append(design_statistics(S, rng)) 

320 keys = list(observed) 

321 p_values[method] = _mc_p_value( 

322 np.array([observed[k] for k in keys]), 

323 np.array([[s[k] for k in keys] for s in sims]), 

324 ) 

325 

326 best = max(p_values, key=p_values.get) 

327 consistent = [m for m in sorted(p_values, key=p_values.get, reverse=True) if p_values[m] >= alpha] 

328 p_txt = ", ".join(f"{m} p={p_values[m]:.3f}" for m in sorted(p_values, key=p_values.get, reverse=True)) 

329 stat_txt = f"strata filled {observed['strata']:.2f}" 

330 if "log_cd" in observed: 

331 stat_txt += f", centered discrepancy {10 ** observed['log_cd']:.2e}" 

332 

333 if not consistent: 

334 method = "Uncertain" 

335 explanation = ( 

336 f"No candidate design matches ({p_txt}; {stat_txt}){note}. " 

337 "Data may be observational or have non-uniform marginals -> Uncertain" 

338 ) 

339 else: 

340 method = best 

341 also = f"; also consistent with {', '.join(consistent[1:])}" if len(consistent) > 1 else "" 

342 explanation = ( 

343 f"Monte Carlo test vs {n_sim} simulated designs ({p_txt}; {stat_txt}){also}{note} -> {method}" 

344 ) 

345 return SamplingDetection( 

346 method=method, 

347 explanation=explanation, 

348 statistics=observed, 

349 p_values=p_values, 

350 n_sim=n_sim, 

351 columns_used=continuous.tolist(), 

352 )
