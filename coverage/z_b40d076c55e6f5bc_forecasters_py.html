2"""ARIMA / SARIMA / LSTM / naive forecasters with one interface (audit item 6). 

3 

4Every forecaster has ``fit(y_train)`` and ``forecast(horizon) -> np.ndarray``, 

5and the forecast is always on the **original scale** of ``y_train``, so all 

6models can be scored against the same held-out values by one metric function 

7(:mod:`multioutreg.time_series.ts_dynamic_fit.src.evaluate`). 

8 

9ARIMA / SARIMA choose ``d`` with the ADF test (at most :data:`MAX_D` 

10differences) and pass it to statsmodels, which integrates the forecast back to 

11levels. ``fit(y, order=...)`` skips the order search and only re-estimates the 

12parameters; rolling-origin evaluation uses that to search the order once, on 

13the first (smallest) training window. 

14""" 

15 

16from __future__ import annotations 

17 

18import itertools 

19import warnings 

20from typing import Optional 

21 

22import numpy as np 

23 

24#: Largest number of differences the ADF loop may apply (audit item 24). 

25MAX_D = 2 

26#: Largest AR / MA order in the ARIMA grid (p, q in 0..7). 

27ARIMA_MAX_ORDER = 7 

28#: Longest season SARIMA models. The state-space fit slows sharply with s: 

29#: one (1,0,1)(1,0,1,s) fit takes ~10 s at s = 52 and ~90 s at s = 123, and the 

30#: search runs 16 of them. Longer seasons (365 for daily data) still set the 

31#: horizon and the seasonal-naive baseline; SARIMA then fits without a season. 

32SARIMA_MAX_SEASON = 52 

33

35def _as_1d(y) -> np.ndarray: 

36 y = np.asarray(y, dtype=float).ravel() 

37 if y.size == 0 or not np.all(np.isfinite(y)): 

38 raise ValueError("the training series must be non-empty and finite") 

39 return y 

40

42def _standardize(y): 

43 """``(y - mean) / sd`` and the (mean, sd) to undo it. 

44 

45 ARIMA is equivariant under affine maps of y, but statsmodels' optimiser is 

46 not: on a series around 1e6 it can converge elsewhere than on the same 

47 series around 1e3. Fitting on the standardised series makes the order 

48 search and forecasts scale-invariant. 

49 """ 

50 mu, sd = float(np.mean(y)), float(np.std(y)) 

51 sd = sd if sd > 0 else 1.0 

52 return (y - mu) / sd, (mu, sd) 

53 

54 

55def choose_d(y, alpha: float = 0.05, max_d: int = MAX_D) -> int: 

56 """Number of differences until the ADF test rejects a unit root (at most ``max_d``).""" 

57 from statsmodels.tsa.stattools import adfuller 

58 

59 d, z = 0, np.asarray(y, dtype=float) 

60 while d < max_d: 

61 if np.ptp(z) == 0: # constant: stationary, and adfuller would fail 

62 break 

63 if adfuller(z)[1] <= alpha: 

64 break 

65 z = np.diff(z) 

66 d += 1 

67 return d

106class ArimaForecaster: 

107 """ARIMA(p, d, q): ``d`` from the ADF test, (p, q) by AIC over 0..7.""" 

108 

109 name = "ARIMA" 

110 

111 def __init__(self, max_order: int = ARIMA_MAX_ORDER, n_jobs: int = 1): 

112 self.max_order = max_order 

113 self.n_jobs = n_jobs 

114 

115 @staticmethod 

116 def _fit_one(y, order): 

117 import statsmodels.api as sm 

118 

119 with warnings.catch_warnings(): 

120 warnings.simplefilter("ignore") 

121 try: 

122 return sm.tsa.ARIMA(y, order=order).fit() 

123 except Exception: 

124 return None 

125 

126 def fit(self, y_train, order=None) -> "ArimaForecaster": 

127 y, self.scale_ = _standardize(_as_1d(y_train)) 

128 if order is None: 

129 d = choose_d(y) 

130 grid = [(p, d, q) for p, q in itertools.product( 

131 range(self.max_order + 1), repeat=2)] 

132 fits = _parallel(self._fit_one, [(y, o) for o in grid], self.n_jobs) 

133 scored = [(r.aic, o, r) for o, r in zip(grid, fits) 

134 if r is not None and np.isfinite(r.aic)] 

135 if not scored: 

136 raise RuntimeError("ARIMA: no (p, d, q) order could be fitted") 

137 _, order, self.results_ = min(scored, key=lambda t: t[0]) 

138 else: 

139 self.results_ = self._fit_one(y, tuple(order)) 

140 if self.results_ is None: 

141 raise RuntimeError(f"ARIMA{tuple(order)} could not be fitted") 

142 self.order_ = tuple(order) 

143 return self 

144 

145 def forecast(self, horizon: int) -> np.ndarray: 

146 mu, sd = self.scale_ 

147 return mu + sd * np.asarray(self.results_.forecast(horizon), dtype=float)

150class SarimaForecaster: 

151 """SARIMA(p, d, q)(P, 0, Q, s) over p, q, P, Q in {0, 1}. 

152 

153 ``d`` comes from the ADF test and ``s`` from the ACF of the differenced 

154 series (or ``seasonality``). Without a season (s < 2, s above 

155 :data:`SARIMA_MAX_SEASON`, or fewer than two seasons plus two points of 

156 data) it reduces to ARIMA(p, d, q), p, q <= 1. 

157 """ 

158 

159 name = "SARIMA" 

160 

161 def __init__(self, seasonality: Optional[int] = None, n_jobs: int = 1): 

162 self.seasonality = seasonality 

163 self.n_jobs = n_jobs 

164 

165 @staticmethod 

166 def _fit_one(y, order, seasonal_order): 

167 import statsmodels.api as sm 

168 

169 with warnings.catch_warnings(): 

170 warnings.simplefilter("ignore") 

171 try: 

172 return sm.tsa.statespace.SARIMAX( 

173 y, order=order, seasonal_order=seasonal_order).fit(disp=False) 

174 except Exception: 

175 return None 

176 

177 def fit(self, y_train, order=None) -> "SarimaForecaster": 

178 y, self.scale_ = _standardize(_as_1d(y_train)) 

179 if order is None: 

180 from multioutreg.time_series.seasonality import detect_seasonal_period 

181 

182 d = choose_d(y) 

183 s = self.seasonality 

184 if s is None: 

185 s = detect_seasonal_period(np.diff(y, n=d) if d else y) or 0 

186 seasonal = 2 <= s <= SARIMA_MAX_SEASON and y.size >= 2 * s + 2 

187 grid = [((p, d, q), (P, 0, Q, s) if seasonal else (0, 0, 0, 0)) 

188 for p in (0, 1) for q in (0, 1) 

189 for P in ((0, 1) if seasonal else (0,)) 

190 for Q in ((0, 1) if seasonal else (0,))] 

191 fits = _parallel(self._fit_one, [(y, o, so) for o, so in grid], self.n_jobs) 

192 scored = [(r.aic, o, r) for o, r in zip(grid, fits) 

193 if r is not None and np.isfinite(r.aic)] 

194 if not scored: 

195 raise RuntimeError("SARIMA: no (p, d, q)(P, D, Q, s) order could be fitted") 

196 _, order, self.results_ = min(scored, key=lambda t: t[0]) 

197 else: 

198 self.results_ = self._fit_one(y, tuple(order[0]), tuple(order[1])) 

199 if self.results_ is None: 

200 raise RuntimeError(f"SARIMA{order} could not be fitted") 

201 self.order_ = (tuple(order[0]), tuple(order[1])) 

202 return self 

203 

204 def forecast(self, horizon: int) -> np.ndarray: 

205 mu, sd = self.scale_ 

206 return mu + sd * np.asarray(self.results_.forecast(horizon), dtype=float)

209class LstmForecaster: 

210 """Direct multi-step LSTM: ``train_window`` past values -> next ``horizon`` values. 

211 

212 The scaler is fitted on the training series only, torch and numpy are 

213 seeded, and training uses shuffled mini-batches (the old per-sample loop 

214 took ~2 minutes for 150 points). 

215 """ 

216 

217 name = "LSTM" 

218 

219 def __init__(self, horizon: int = 1, train_window: int = 10, epochs: int = 200, 

220 batch_size: int = 16, lr: float = 0.005, random_state: int = 0): 

221 self.horizon = horizon 

222 self.train_window = train_window 

223 self.epochs = epochs 

224 self.batch_size = batch_size 

225 self.lr = lr 

226 self.random_state = random_state 

227 

228 def fit(self, y_train, order=None) -> "LstmForecaster": 

229 import torch 

230 from torch import nn 

231 from sklearn.preprocessing import MinMaxScaler 

232 

233 from multioutreg.time_series.ts_dynamic_fit.algs.lstm import LSTMModel 

234 

235 y = _as_1d(y_train) 

236 w, h = self.train_window, self.horizon 

237 n_seq = y.size - w - h + 1 

238 if n_seq < 1: 

239 raise ValueError( 

240 f"LSTM needs at least train_window + horizon = {w + h} training points, " 

241 f"got {y.size}") 

242 torch.manual_seed(self.random_state) 

243 rng = np.random.default_rng(self.random_state) 

244 self.scaler_ = MinMaxScaler(feature_range=(-1, 1)).fit(y.reshape(-1, 1)) 

245 z = self.scaler_.transform(y.reshape(-1, 1)).ravel() 

246 X = torch.tensor(np.stack([z[i:i + w] for i in range(n_seq)]), dtype=torch.float32) 

247 Y = torch.tensor(np.stack([z[i + w:i + w + h] for i in range(n_seq)]), dtype=torch.float32) 

248 

249 model = LSTMModel(output_size=h) 

250 opt = torch.optim.Adam(model.parameters(), lr=self.lr) 

251 loss_fn = nn.MSELoss() 

252 model.train() 

253 for _ in range(self.epochs): 

254 perm = rng.permutation(n_seq) 

255 for start in range(0, n_seq, self.batch_size): 

256 idx = perm[start:start + self.batch_size] 

257 opt.zero_grad() 

258 loss = loss_fn(model(X[idx]), Y[idx]) 

259 loss.backward() 

260 opt.step() 

261 model.eval() 

262 self.model_ = model 

263 self.context_ = z[-w:] 

264 return self 

265 

266 def forecast(self, horizon: int) -> np.ndarray: 

267 import torch 

268 

269 if horizon != self.horizon: 

270 raise ValueError( 

271 f"this LSTM was trained for horizon={self.horizon}, asked for {horizon}") 

272 with torch.no_grad(): 

273 out = self.model_(torch.tensor(self.context_, dtype=torch.float32)).numpy() 

274 return self.scaler_.inverse_transform(out.reshape(-1, 1)).ravel()

277def _parallel(fn, arg_list, n_jobs): 

278 if n_jobs == 1: 

279 return [fn(*args) for args in arg_list] 

280 from joblib import Parallel, delayed 

281 

282 return Parallel(n_jobs=n_jobs)(delayed(fn)(*args) for args in arg_list)
